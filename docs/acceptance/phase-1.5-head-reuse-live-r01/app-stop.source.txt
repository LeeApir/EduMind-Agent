from pathlib import Path
from uuid import UUID
from sqlalchemy import select,func
from sqlalchemy.ext.asyncio import async_sessionmaker
import asyncio,json
from scripts import head_course_goal as entry
from app.services.free_course_snapshots import SnapshotRepository,WriteFence
from app.services.free_course_scheduler import Scheduler,Lease
from app.models.free_course_planning import FreeCoursePlanningResult
from app.models.free_scene_generation import FreeSceneGenerationResult
owner=UUID("0b1b71df-9b00-470e-abc4-ae41aeb20cf9")
cid=UUID("6be90c82-adfd-408d-82b6-32125db0dcbf")
oid=UUID("ecb65468-da85-4772-8a5d-b7f1bb5e8fa1")
async def finish(db):
 repo=SnapshotRepository(db,owner)
 op=await repo._operation(cid,oid)
 lease=Lease(owner,cid,WriteFence(oid,op.lease_epoch,UUID(int=0)))
 scheduler=Scheduler(async_sessionmaker(db.bind,expire_on_commit=False))
 repo,course,op=await scheduler._locked(db,lease,live=False)
 assert op.state in ("queued","running")
 await scheduler._terminal(db,lease,op,"failed","TEACHING_REVIEW_FAILED")
 await db.flush()
 return dict(owner=str(owner),course_id=str(cid),operation_id=str(oid),state=op.state,reason_code=op.reason_code,head_version=(await repo.get_course(cid))["head_version"],planning_results=await db.scalar(select(func.count()).select_from(FreeCoursePlanningResult).where(FreeCoursePlanningResult.course_id==cid)),scene_generation_results=await db.scalar(select(func.count()).select_from(FreeSceneGenerationResult).where(FreeSceneGenerationResult.course_id==cid)),new_execution_lease_acquired=False,provider_calls_in_cleanup=0)
result=asyncio.run(entry.database(finish))
Path("docs/acceptance/phase-1.5-head-reuse-live-r01/app-stop-result.json").write_text(json.dumps(result,indent=2)+"\n")
print(json.dumps(result))
