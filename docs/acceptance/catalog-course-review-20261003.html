数组

array

讲解

# 数组

数组把同类型元素放在连续空间中。有效下标是0到n-1；长度n和容量cap不是同一个数。知道下标时访问是O(1)。中间插入需要搬移后面的元素，最坏O(n)；末尾插入只有在尚有容量且不扩容时才是O(1)。

示例先检查容量和插入位置，再从后向前移动，避免覆盖尚未搬移的值。空数组只能在0处插入，满数组拒绝插入。示例展示固定容量，不承诺自动扩容。
完整 C 示例（含边界断言）
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdio.h>
static bool insert(int *a, size_t *n, size_t cap, size_t pos, int value) {
    if (*n >= cap || pos > *n) return false;
    for (size_t i = *n; i > pos; --i) a[i] = a[i - 1];
    a[pos] = value;
    ++*n;
    return true;
}
int main(void) {
    int a[3] = {0}; size_t n = 0;
    assert(!insert(a, &n, 3, 1, 7));
    assert(insert(a, &n, 3, 0, 1));
    assert(insert(a, &n, 3, 1, 2));
    assert(insert(a, &n, 3, 1, 9));
    assert(!insert(a, &n, 3, 0, 8));
    for (size_t i = 0; i < n; ++i) printf("%d%s", a[i], i + 1 == n ? "\n" : " ");
    return 0;
}

C 指针

c-pointer

讲解

# C 指针

指针变量保存地址，&value取得对象的地址，*p在p有效时访问所指对象。指针变量、它保存的地址、该地址中的值是不同概念。不能解引用未初始化指针或NULL。

示例p指向main中的value，该对象的生存期覆盖全部使用过程。修改*p会修改value。自动对象不能free；malloc获得的对象由拥有者在使用完后free。释放后的指针不能继续解引用，函数不能返回局部自动对象的地址供调用者使用。
完整 C 示例（含边界断言）
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdio.h>
int main(void) {
    int value = 4; int *p = &value;
    assert(p != NULL);
    printf("%d\n", *p);
    *p = 5;
    assert(value == 5);
    int *empty = NULL;
    if (empty != NULL) puts("This branch is not reached.");
    printf("%d\n", value);
    return 0;
}

链表概念

linked-list-concept

讲解

# 链表概念

链表通过链接组织节点，节点包含数据域和指向后继的指针域。节点不必在内存中连续，next保存后继地址而不是后继数据值。无头哨兵的空链表可用head=NULL表示；普通非循环链表的最后一个next为NULL。

示例使用两个生存期覆盖main的自动节点，用next链接起来，不需要malloc或free。采用动态分配时应另定所有权：分配失败不能覆盖原head，释放节点前先保存后继。已知前驱后的链接修改为O(1)，寻找位置仍可能O(n)。
完整 C 示例（含边界断言）
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdio.h>
typedef struct Node { int value; struct Node *next; } Node;
int main(void) {
    Node second = {2, NULL}; Node first = {1, &second};
    Node *head = &first;
    assert(head->next == &second && second.next == NULL);
    printf("%d %d\n", head->value, head->next->value);
    /* Both nodes are automatic objects, valid until main returns. Do not free them. */
    return 0;
}

单链表

single-linked-list

讲解

# 单链表

单链表每个节点只有一个next，只能沿next前进。本文约定没有头哨兵，head直接指向首个数据节点；head=NULL表示空表。其他教材若使用头哨兵，判断空表的表达式会不同，不能混用。

示例insert_at接收Node **，仅分配成功后修改链接，失败返回false而保留旧表。main检查每次返回值，失败时释放仍拥有的全部节点。禁止head=某个失败会返回NULL的插入函数这样的无条件覆盖。free_list先保存next再free当前节点，不能释放后读取head->next。
完整 C 示例（含边界断言）
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#ifndef EDUMIND_ALLOC_FAIL_AFTER
#define EDUMIND_ALLOC_FAIL_AFTER -1
#endif
typedef struct Node { int value; struct Node *next; } Node;
static Node *make_node(int value) {
    static int allocations = 0;
    if (EDUMIND_ALLOC_FAIL_AFTER >= 0 && allocations >= EDUMIND_ALLOC_FAIL_AFTER) return NULL;
    Node *p = malloc(sizeof *p);
    if (p == NULL) return NULL;
    ++allocations; p->value = value; p->next = NULL;
    return p;
}
static bool insert_at(Node **head, size_t pos, int value) {
    Node **link = head;
    while (pos > 0) {
        if (*link == NULL) return false;
        link = &(*link)->next; --pos;
    }
    Node *node = make_node(value);
    if (node == NULL) return false; /* Existing head/links remain untouched. */
    node->next = *link;
    *link = node;
    return true;
}
static void free_list(Node *head) {
    while (head != NULL) { Node *next = head->next; free(head); head = next; }
}
static void print_list(const Node *head) {
    while (head != NULL) { printf("%d%s", head->value, head->next == NULL ? "\n" : " "); head = head->next; }
}
int main(void) {
    Node *head = NULL;
    assert(!insert_at(&head, 1, 99));
    if (!insert_at(&head, 0, 2) || !insert_at(&head, 0, 1)) {
        free_list(head); puts("allocation unavailable"); return 0;
    }
    assert(head != NULL && head->next != NULL && head->next->next == NULL);
    print_list(head); free_list(head); return 0;
}

链表遍历

linked-list-traversal

讲解

# 链表遍历

从head开始，每次处理当前节点，再执行cur=cur->next，直到cur==NULL。循环条件是当前指针是否为NULL，不能用cur->next!=NULL作为普通遍历条件，否则会漏掉末节点并可能对空表解引用。

示例只遍历合法非循环链表；循环或悬空指针输入不在此函数前提内。访问所有n个节点是O(n)时间、O(1)额外空间。链表按位置访问需要从头前进，不能像数组一样直接计算任意节点地址。
完整 C 示例（含边界断言）
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdio.h>
typedef struct Node { int value; struct Node *next; } Node;
static size_t length(const Node *head) {
    size_t count = 0;
    for (const Node *cur = head; cur != NULL; cur = cur->next) ++count;
    return count;
}
int main(void) {
    assert(length(NULL) == 0);
    Node third = {3, NULL}; Node second = {2, &third}; Node first = {1, &second};
    assert(length(&first) == 3);
    for (const Node *cur = &first; cur != NULL; cur = cur->next) printf("%d\n", cur->value);
    return 0;
}

链表插入

linked-list-insertion

讲解

# 链表插入

插入位置pos采用从0开始的间隙编号，允许0到当前长度；0为头插，长度为尾插，超过长度非法。示例用指向链接的指针Node **统一处理头部和中间位置。

找到需要替换的链接后，先分配节点。失败返回false，原表不变；成功后先node->next=*link保存旧后继，再*link=node连入新节点。先覆盖旧链接而未保存后继会丢失原链。已知前驱时链接修改O(1)，按位置寻找需要O(n)。动画展示相同链接顺序，不执行用户代码。
完整 C 示例（含边界断言）

#include <stdbool.h>
#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>

#define EDUMIND_ALLOC_FAIL_AFTER -1
#endif

    Node *p = malloc(sizeof *p);
    if (p == NULL) return NULL;

    return p;
}

        link = &(*link)->next; --pos;
    }
    Node *node = make_node(value);
    if (node == NULL) return false; /* Existing head/links remain untouched. */

    *link = node;

    Node *saved = head;
    if (!insert_at(&head, 1, 2)) {
        assert(head == saved && head->next->value == 3);
        free_list(head); puts("allocation unavailable"); return 0;

    assert(!insert_at(&head, 4, 99));
    print_list(head); free_list(head); return 0;

链表删除

linked-list-deletion

讲解

# 链表删除

删除位置pos从0开始，只允许0到长度-1。必须确定目标存在；空表或超出范围时返回false且保持原表。无头哨兵的头删必须能更新调用者head。

示例找到指向目标的链接，先保存target，再把该链接改为target->next，最后free(target)。释放之后不再读取target。main最后删除首节点与唯一剩余节点，head回到NULL。寻找位置O(n)，已知前驱时修改链接和释放操作不包含搜索开销。
完整 C 示例（含边界断言）
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#ifndef EDUMIND_ALLOC_FAIL_AFTER
#define EDUMIND_ALLOC_FAIL_AFTER -1
#endif
typedef struct Node { int value; struct Node *next; } Node;
static Node *make_node(int value) {
    static int allocations = 0;
    if (EDUMIND_ALLOC_FAIL_AFTER >= 0 && allocations >= EDUMIND_ALLOC_FAIL_AFTER) return NULL;
    Node *p = malloc(sizeof *p);
    if (p == NULL) return NULL;
    ++allocations; p->value = value; p->next = NULL;
    return p;
}
static bool insert_at(Node **head, size_t pos, int value) {
    Node **link = head;
    while (pos > 0) {
        if (*link == NULL) return false;
        link = &(*link)->next; --pos;
    }
    Node *node = make_node(value);
    if (node == NULL) return false; /* Existing head/links remain untouched. */
    node->next = *link;
    *link = node;
    return true;
}
static void free_list(Node *head) {
    while (head != NULL) { Node *next = head->next; free(head); head = next; }
}
static void print_list(const Node *head) {
    while (head != NULL) { printf("%d%s", head->value, head->next == NULL ? "\n" : " "); head = head->next; }
}
static bool delete_at(Node **head, size_t pos) {
    Node **link = head;
    while (pos > 0) {
        if (*link == NULL) return false;
        link = &(*link)->next; --pos;
    }
    Node *target = *link;
    if (target == NULL) return false;
    *link = target->next; free(target); return true;
}
int main(void) {
    Node *head = NULL; assert(!delete_at(&head, 0));
    if (!insert_at(&head, 0, 1) || !insert_at(&head, 1, 2) || !insert_at(&head, 2, 3)) {
        free_list(head); puts("allocation unavailable"); return 0;
    }
    assert(!delete_at(&head, 3)); assert(delete_at(&head, 1));
    print_list(head);
    assert(delete_at(&head, 0)); assert(delete_at(&head, 0));
    assert(head == NULL && !delete_at(&head, 0));
    free_list(head); return 0;
}

栈

stack

讲解

# 栈

栈遵循后进先出LIFO：push入栈，pop取出并移除栈顶，peek只查看栈顶而不移除。本文数组实现用count表示元素个数，非空时栈顶下标为count-1。

push必须先检查count==CAP，peek/pop必须先检查count==0。不能在空栈中读取data[count-1]，否则下标下溢或越界。操作返回bool表示成功，输出参数仅在成功时写入。此固定容量实现三种操作均为O(1)，没有扩容。
完整 C 示例（含边界断言）
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdio.h>
enum { CAP = 2 };
typedef struct { int data[CAP]; size_t count; } Stack;
static bool push(Stack *s, int v) {
    if (s->count == CAP) return false;
    s->data[s->count++] = v; return true;
}
static bool peek(const Stack *s, int *out) {
    if (s->count == 0) return false;
    *out = s->data[s->count - 1]; return true;
}
static bool pop(Stack *s, int *out) {
    if (s->count == 0) return false;
    *out = s->data[--s->count]; return true;
}
int main(void) {
    Stack s = {{0}, 0}; int value = -1;
    assert(!peek(&s, &value) && value == -1); assert(!pop(&s, &value));
    assert(push(&s, 1)); assert(push(&s, 2)); assert(!push(&s, 3));
    assert(peek(&s, &value) && value == 2);
    assert(pop(&s, &value)); printf("%d\n", value);
    assert(pop(&s, &value)); printf("%d\n", value);
    assert(!pop(&s, &value) && !peek(&s, &value)); return 0;
}

队列

queue

讲解

# 队列

队列遵循先进先出FIFO：enqueue从队尾入队，dequeue从队首移除并返回，peek只看队首。抽象顺序与具体实现复杂度要分开。

示例使用连续数组，入队加到末尾，出队后整体左移，因此enqueue/peek为O(1)、dequeue最坏O(n)。空/满必须在访问前检查，失败不修改输出参数。若使用循环数组可避免搬移，但需正确区分空与满，见循环队列节点。
完整 C 示例（含边界断言）
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdio.h>
enum { CAP = 3 };
typedef struct { int data[CAP]; size_t count; } Queue;
static bool enqueue(Queue *q, int value) {
    if (q->count == CAP) return false;
    q->data[q->count++] = value; return true;
}
static bool peek(const Queue *q, int *out) {
    if (q->count == 0) return false;
    *out = q->data[0]; return true;
}
static bool dequeue(Queue *q, int *out) {
    if (q->count == 0) return false;
    *out = q->data[0];
    for (size_t i = 1; i < q->count; ++i) q->data[i - 1] = q->data[i];
    --q->count; return true;
}
int main(void) {
    Queue q = {{0}, 0}; int value = -1;
    assert(!peek(&q, &value) && value == -1); assert(!dequeue(&q, &value));
    assert(enqueue(&q, 1)); assert(enqueue(&q, 2)); assert(enqueue(&q, 3));
    assert(!enqueue(&q, 4)); assert(peek(&q, &value) && value == 1);
    for (int i = 1; i <= 3; ++i) { assert(dequeue(&q, &value) && value == i); printf("%d\n", value); }
    assert(!dequeue(&q, &value)); return 0;
}

# 循环队列

循环队列使用模运算使数组下标环绕，避免线性数组出队后的整体搬移。本文CAP是实际容量，front指向下次出队位置，rear指向下次入队位置，另用count记录元素个数。

空判定count==0，满判定count==CAP。front==rear可能是空也可能是满，不能单独判断。另一种“预留一个槽”的设计有CAP-1有效容量，判满式(rear+1)%CAP==front；不能把该式混入本文count方案。示例覆盖满队拒绝、出队后重新入队和多次环绕，三个操作均为O(1)。
完整 C 示例（含边界断言）
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdio.h>
enum { CAP = 3 };
typedef struct { int data[CAP]; size_t front, rear, count; } Queue;
static bool enqueue(Queue *q, int value) {
    if (q->count == CAP) return false;
    q->data[q->rear] = value; q->rear = (q->rear + 1) % CAP; ++q->count; return true;
}
static bool peek(const Queue *q, int *out) {
    if (q->count == 0) return false;
    *out = q->data[q->front]; return true;
}
static bool dequeue(Queue *q, int *out) {
    if (q->count == 0) return false;
    *out = q->data[q->front]; q->front = (q->front + 1) % CAP; --q->count; return true;
}
int main(void) {
    Queue q = {{0}, 0, 0, 0}; int value = -1;
    assert(!peek(&q, &value) && value == -1); assert(!dequeue(&q, &value));
    assert(enqueue(&q, 1)); assert(enqueue(&q, 2)); assert(enqueue(&q, 3));
    assert(q.front == q.rear && q.count == CAP); assert(!enqueue(&q, 4));
    assert(dequeue(&q, &value) && value == 1); assert(enqueue(&q, 4));
    for (int i = 2; i <= 4; ++i) { assert(dequeue(&q, &value) && value == i); printf("%d\n", value); }
    assert(q.count == 0 && q.front == q.rear); assert(!dequeue(&q, &value));
    return 0;
}

数组 vs 链表（预设演示）

性能视角

数组按下标访问O(1)，连续存储通常有局部性优势。单链表按位置寻找O(n)；只有已知前驱时插删的链接修改才是O(1)，寻找前驱与分配开销不能省略。

内存视角

数组需要连续空间；动态数组扩容可能搬移内容。链表节点可分散分配，但每节点多一个next且有分配器开销；malloc失败与所有权必须处理。不能简单断言链表总更省内存。

使用视角

频繁随机访问通常更适合数组；已知节点/前驱且频繁插删时可以考虑链表。选择还取决于规模、分配成本和接口需求，本演示不针对任意场景生成推荐。
