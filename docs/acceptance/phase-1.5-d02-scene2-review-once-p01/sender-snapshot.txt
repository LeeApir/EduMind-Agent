"""One exact saved D02 scene review. Never resumes r05 or changes its verdict."""

# ruff: noqa: E402
import argparse
import asyncio
import hashlib
import json
import sys
import time
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path[:0] = [str(ROOT / "backend"), str(ROOT / "scripts")]
import httpx
from jsonschema import Draft202012Validator
from acceptance_support.p15_candidate_compatibility import evidence, redact_check
from app.services.deepseek_responses import _wire_target
from isolated_diagnostics import p15_review_observability as compat
from isolated_diagnostics.p15_course_sol_bridge import review_request
from run_p15_sol_compatibility_once import credential, target

BASE = ROOT / "docs/acceptance/phase-1.5-d02-scene2-review-once-p01"
OUTPUT = Path("/tmp/edumind-p15-d02-scene2-review-once-p01")
SOURCE_SHA = "23bba98d9b6a45093cb3feb42d3e8ee6f3781ab4580f3de82473297c69b20e30"


def gate(execute=False):
    raw = (BASE / "manifest.json").read_bytes()
    manifest = json.loads(raw)
    for name, sha in manifest["sha256"].items():
        path = (ROOT / name).resolve()
        if not path.is_relative_to(ROOT) or hashlib.sha256(path.read_bytes()).hexdigest() != sha:
            raise ValueError("FROZEN_PACKAGE_CHANGED")
    request = json.loads((BASE / "request.json").read_bytes())
    payload = json.loads(request["input"][-1]["content"])
    if payload["source_sha256"] != SOURCE_SHA or request != review_request(
        "development_scene_review", payload
    ):
        raise ValueError("EXACT_ORIGINAL_SCENE_REQUEST_REQUIRED")
    grant = json.loads((BASE / "execution-approval.json").read_bytes())
    package = hashlib.sha256(raw).hexdigest()
    if execute and (
        grant.get("approved") is not True or grant.get("package_sha256") != package
        or grant.get("requests") != 1 or grant.get("wall_seconds") != 120
        or grant.get("synthetic_transfer_approved") is not True
        or grant.get("unknown_fee_and_token_exception_approved") is not True
        or not grant.get("evidence_ref") or not grant.get("human_message")
    ):
        raise ValueError("EXACT_ONE_REQUEST_AND_FEE_AUTHORITY_REQUIRED")
    return package, request


def evaluate(body, response):
    payload = json.loads(body["input"][-1]["content"])
    rejection = None
    try:
        compat.ReviewObservationAdapter.validate_scene(body, response, payload)
    except (ValueError, TypeError, KeyError):
        # Only reproduce a bounded code, never echo arbitrary upstream text in an exception.
        try:
            compat.ReviewObservationAdapter.validate_scene(body, response, payload)
        except ValueError as exc:
            code = str(exc)
            rejection = code if code.isupper() and code.replace("_", "").isalnum() else "PROTOCOL_FAILED"
        except (TypeError, KeyError):
            rejection = "PROTOCOL_SHAPE_FAILED"
    opinion = None
    try:
        messages = [i for i in response["output"] if i.get("type") == "message"]
        if (response.get("model") != "gpt-6.1-sol" or response.get("status") != "completed"
            or response.get("tools") not in (None, []) or len(messages) != 1
            or messages[0].get("role") != "assistant"
            or len(messages[0]["content"]) != 1
            or messages[0]["content"][0].get("type") != "output_text"):
            raise ValueError("FINAL_MESSAGE_INVALID")
        final = json.loads(messages[0]["content"][0]["text"])
        Draft202012Validator(compat.schema()).validate(final)
        compat.scene.original.validate_opinion(payload, final)
        opinion = final
    except (ValueError, TypeError, KeyError, IndexError):
        pass
    return dict(complete_protocol_pass=rejection is None, protocol_rejection=rejection,
                source_bound_final_opinion=opinion, final_opinion_only_is_not_protocol_pass=True,
                T038_completed=False, independent_human_truth=False)


async def run_once(package, body, key, *, output=OUTPUT, target_factory=target,
                   transport_factory=lambda: httpx.AsyncHTTPTransport(retries=0, trust_env=False)):
    raw = json.dumps(body, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode()
    redact_check(raw, key)
    if len(raw) > 131072:
        raise ValueError("REQUEST_TOO_LARGE")
    output = Path(output)
    output.mkdir(mode=0o700, exist_ok=False)
    started, deadline = time.monotonic(), time.time() + 120
    evidence(output / "reservation.json", dict(package_sha256=package, requests=1,
        deadline_epoch=deadline, actual_fee=None, token_bounds_verified=False))
    sent, status, response, reason = False, None, None, None
    try:
        async with asyncio.timeout(119):
            approved = await asyncio.to_thread(target_factory)
            if (approved.url != "https://aiapi.world/v1" or approved.hostname != "aiapi.world"
                or approved.port != 443):
                raise ValueError("APPROVED_TARGET_CHANGED")
            url, host = _wire_target(approved)
            evidence(output / "request.json", raw)
            async with httpx.AsyncClient(transport=transport_factory(), trust_env=False,
                follow_redirects=False, timeout=110) as client:
                request = client.build_request("POST", url, content=raw, headers={
                    "Host": host, "Content-Type": "application/json", "Authorization": "Bearer " + key})
                request.extensions["sni_hostname"] = approved.hostname
                evidence(output / "dispatch.json", dict(url="https://aiapi.world/v1/responses",
                    request_sha256=hashlib.sha256(raw).hexdigest(), no_retry=True, attempt=1))
                sent = True
                res = await client.send(request, stream=True, follow_redirects=False)
                try:
                    status = res.status_code
                    evidence(output / "status.json", dict(http_status=status))
                    data = bytearray()
                    async for chunk in res.aiter_bytes():
                        data.extend(chunk)
                        if len(data) > 2097152:
                            raise ValueError("RESPONSE_TOO_LARGE")
                finally:
                    await res.aclose()
            data = bytes(data)
            redact_check(data, key)
            evidence(output / "response.json", data)
            response = json.loads(data)
            if status != 200:
                raise ValueError("HTTP_NOT_200")
    except BaseException as exc:
        code = str(exc)
        reason = code if isinstance(exc, ValueError) and code.isupper() and code.replace(
            "_", "").isalnum() else "TRANSPORT_TIMEOUT_OR_RESPONSE_FAILURE"
    result = dict(package_sha256=package, source_sha256=SOURCE_SHA, dispatches=int(sent),
        http_status=status, safe_failure=reason, elapsed_seconds=round(time.monotonic()-started, 3),
        actual_fee=None, r05_resumed=False, T038_completed=False, reported_usage=None, observation=None)
    if response is not None and reason is None:
        result["observation"] = evaluate(body, response)
        usage = response.get("usage")
        if isinstance(usage, dict):
            result["reported_usage"] = {k: usage.get(k) for k in (
                "input_tokens", "output_tokens", "total_tokens")}
    result["settlement"] = "observed_response" if result["observation"] and result[
        "observation"]["complete_protocol_pass"] else "unknown_preserved"
    evidence(output / "summary.json", result)
    return result


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group()
    mode.add_argument("--check", action="store_true")
    mode.add_argument("--execute", action="store_true")
    args = parser.parse_args()
    package, body = gate(execute=args.execute)
    if not args.execute:
        print(json.dumps(dict(package_sha256=package, requests=1, real_calls=0)))
        return
    result = asyncio.run(run_once(package, body, credential()))
    print(json.dumps(result, ensure_ascii=False))


if __name__ == "__main__":
    main()
